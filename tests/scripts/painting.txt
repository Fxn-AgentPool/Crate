# Mask paint: painting black into a white mask hides the layer there.
mode live
tool paint
add-fill
rename Red
set base_color=#C03020
add-mask
add-mask-paint
render target/tests/paint/m_before.png
brush-values mask=0
brush size=80 hardness=0.8 flow=1
stroke 0.3,0.5 0.7,0.5
render target/tests/paint/m_black.png
expect-different target/tests/paint/m_before.png target/tests/paint/m_black.png
# ...and white paints it back (soft edges blend, so compare with the black result).
brush-values mask=1
stroke 0.3,0.5 0.7,0.5
render target/tests/paint/m_white.png
expect-different target/tests/paint/m_black.png target/tests/paint/m_white.png
