new assets/crate.glb 1024
bake ao curvature
size 640x480
camera 35 22 1.3
add-fill
set base_color=#2A4C7D roughness=0.6
add-fill
rename Spots
set base_color=#E0C040
# Mask: cellular noise, sharpened, then blurred (a cached filter).
node cells cellular
node-set cells scale=12
node sharp levels
node-set sharp in_black=0.15 in_white=0.2
link cells.out sharp.in
layer-mask sharp
render target/tests/cache/a_sharp.png
node soft blur
node-set soft radius=12
link sharp.out soft.in
layer-mask soft
caches
render target/tests/cache/b_blurred.png
# A cached plain node gives the same image as uncached.
cache cells on
caches
render target/tests/cache/c_cached_cells.png
expect-same target/tests/cache/b_blurred.png target/tests/cache/c_cached_cells.png
# Height to normal: noise height shown as normals.
add-fill
rename Bumps
set base_color=#808080
node bumps noise
node-set bumps scale=20
node hn height-to-normal
node-set hn strength=4
link bumps.out hn.in
node mat material-combine
link hn.out mat.normal
layer-content mat
caches
view normal
render target/tests/cache/d_height_normal.png
