# Geo Fill (docs/reference/brushes-tools-stencils.md#geo-fill) on tests/data/geo.obj: four unit
# quads in a row, facing the camera (see make_geo.py). At camera 0 0 2 and size 400x200 their
# centers are at x = 0.19, 0.40, 0.60, 0.81. Colors are checked in the UV view, centered on a
# point of each quad:
#   Q1 (0.25, 0.5), its two triangles at (0.325, 0.425) and (0.175, 0.575); Q2 (0.55, 0.5);
#   Q3 (1.25, 0.5); Q4 (1.7, 0.5); padding 4 texels below Q1's lower-right triangle at (0.325, 0.346).
# On islands the UV view shows some colors a few levels off their value (a Fill layer of #2040C0
# shows 33,62,185), so colors are checked within 8 levels; the test colors are far apart.
new tests/data/geo.obj 1024
size 400x200
add-fill
set base_color=#808080
add-paint
tool geo-fill
view base_color
# Before any fill: the gray of the fill layer, as the view shows it.
uv-view 0.55 0.5 40
render target/tests/geo_fill/before_q2.png
uv-view 1.25 0.5 40
render target/tests/geo_fill/before_q3.png

# Face: one triangle. Q1's lower-right triangle is at world (-1.25, -0.25), screen
# (0.241, 0.604); its other triangle stays unfilled.
uv-view 0.175 0.575 40
render target/tests/geo_fill/before_q1b.png
camera 0 0 2
fill-by face
brush-values base_color=#C03020
geo-fill 0.241,0.604
uv-view 0.325 0.425 40
render target/tests/geo_fill/face_a.png
expect-color target/tests/geo_fill/face_a.png 192,48,32 8
uv-view 0.175 0.575 40
render target/tests/geo_fill/face_b.png
expect-same target/tests/geo_fill/before_q1b.png target/tests/geo_fill/face_b.png 0
uv-view 0.55 0.5 40
render target/tests/geo_fill/face_q2.png
expect-same target/tests/geo_fill/before_q2.png target/tests/geo_fill/face_q2.png 0
# Padding is filled with its island texels.
uv-view 0.325 0.346 40
render target/tests/geo_fill/face_padding.png
expect-color target/tests/geo_fill/face_padding.png 192,48,32 8

# UV island: Q1 and Q2 share an edge in UV space.
camera 0 0 2
fill-by island
brush-values base_color=#2040C0
geo-fill 0.40,0.5
uv-view 0.25 0.5 40
render target/tests/geo_fill/island_q1.png
expect-color target/tests/geo_fill/island_q1.png 32,64,192 8
uv-view 1.25 0.5 40
render target/tests/geo_fill/island_q3.png
expect-same target/tests/geo_fill/before_q3.png target/tests/geo_fill/island_q3.png 0

# Part: Q3 and Q4, though they are separate islands in tile 1002.
camera 0 0 2
fill-by part
brush-values base_color=#30A040
geo-fill 0.60,0.5
uv-view 1.7 0.5 40
render target/tests/geo_fill/part_q4.png
expect-color target/tests/geo_fill/part_q4.png 48,160,64 8
uv-view 0.55 0.5 40
render target/tests/geo_fill/part_q2.png
expect-color target/tests/geo_fill/part_q2.png 32,64,192 8

# Material: Red is Q1 to Q3, across both parts; Q4 is Blue.
camera 0 0 2
fill-by material
brush-values base_color=#B0B0B0
geo-fill 0.19,0.5
uv-view 1.25 0.5 40
render target/tests/geo_fill/material_q3.png
expect-color target/tests/geo_fill/material_q3.png 176,176,176 8
uv-view 1.7 0.5 40
render target/tests/geo_fill/material_q4.png
expect-color target/tests/geo_fill/material_q4.png 48,160,64 8

# Opacity: half coverage over the light gray.
camera 0 0 2
fill-by face
brush opacity=0.5
brush-values base_color=#000000
geo-fill 0.81,0.5
brush opacity=1
uv-view 1.7 0.5 40
render target/tests/geo_fill/opacity.png
expect-different target/tests/geo_fill/part_q4.png target/tests/geo_fill/opacity.png 10

# UDIM tile, from the UV view: everything in tile 1002, clicked between its islands.
uv-view 1.5 0.5 1.2
fill-by tile
brush-values base_color=#C0A020
geo-fill 0.5,0.5
uv-view 1.25 0.5 40
render target/tests/geo_fill/tile_q3.png
expect-color target/tests/geo_fill/tile_q3.png 192,160,32 8
uv-view 0.55 0.5 40
render target/tests/geo_fill/tile_q2.png
# Tile 1001 keeps the Material fill.
expect-color target/tests/geo_fill/tile_q2.png 176,176,176 8

# Each click is one undo step.
undo
uv-view 1.25 0.5 40
render target/tests/geo_fill/undone.png
expect-color target/tests/geo_fill/undone.png 176,176,176 8

# Nothing under the pointer.
camera 0 0 2
expect-error geo-fill 0.02,0.1
uv-view 5.5 5.5 1
fill-by tile
expect-error geo-fill 0.5,0.5

# A paint mask: Mask value 0 hides the layer in that face.
camera 0 0 2
add-paint
brush-values base_color=#C03020
fill-by part
geo-fill 0.19,0.5
add-mask
add-mask-paint
fill-by face
brush-values mask=0
# Q2's lower-right triangle: world (-0.25, -0.25), screen (0.448, 0.604), UV (0.625, 0.425).
geo-fill 0.448,0.604
uv-view 0.25 0.5 40
render target/tests/geo_fill/mask_q1.png
expect-color target/tests/geo_fill/mask_q1.png 192,48,32 8
uv-view 0.625 0.425 40
render target/tests/geo_fill/mask_q2.png
expect-color target/tests/geo_fill/mask_q2.png 176,176,176 8
